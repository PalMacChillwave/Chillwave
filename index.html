        // --- INITIALIZATION ---

        function init() {

            audio.volume = volume;

            loadStation(0);

            loadTrack(0);

            renderStations();

            renderPlaylist();

            

            // Audio Event Listeners

            audio.addEventListener('timeupdate', updateProgress);

            audio.addEventListener('loadedmetadata', updateProgress);

            audio.addEventListener('ended', playNext);

            

            // Error listener for broken links

            audio.addEventListener('error', (e) => {

                showToast("Cannot load the audio file. The link might be broken or blocked by the host.");

                isPlaying = false;

                updatePlayStateUI();

            });

        }


        // --- CORE FUNCTIONS ---

        function showToast(message) {

            const toast = document.getElementById('toast');

            const toastMessage = document.getElementById('toast-message');

            

            if (message) toastMessage.textContent = message;

            

            clearTimeout(toastTimeout);

            toast.classList.remove('translate-x-[150%]', 'opacity-0');

            

            toastTimeout = setTimeout(() => {

                toast.classList.add('translate-x-[150%]', 'opacity-0');

            }, 6000);

        }


        function attemptPlay() {

            const playPromise = audio.play();

            

            if (playPromise !== undefined) {

                playPromise.then(() => {

                    // Automatic playback started successfully

                    isPlaying = true;

                    updatePlayStateUI();

                }).catch(error => {

                    // Auto-play was prevented

                    console.error("Audio playback prevented:", error);

                    isPlaying = false;

                    updatePlayStateUI();

                    showToast();

                });

            }

        }


        function loadStation(idx) {

            currentStationIdx = idx;

            const station = STATIONS[idx];

            bgImage.style.backgroundImage = `url(${station.bg})`;

            bgGradient.className = `absolute inset-0 z-0 opacity-80 mix-blend-multiply transition-all duration-1000 ${station.color}`;

            renderStations(); 

        }


        function loadTrack(idx) {

            currentTrackIdx = idx;

            const track = TRACKS[idx];

            

            audio.src = track.url;

            currentCover.src = track.cover;

            currentTitle.textContent = track.title;

            currentArtist.textContent = track.artist;

            

            if (isPlaying) {

                attemptPlay();

            }

            renderPlaylist(); 

        }


        function togglePlay() {

            if (isPlaying) {

                audio.pause();

                isPlaying = false;

                updatePlayStateUI();

            } else {

                attemptPlay();

            }

        }


        function playNext() {

            let nextIdx = (currentTrackIdx + 1) % TRACKS.length;

            isPlaying = true; 

            loadTrack(nextIdx);

        }


        function playPrev() {

            let prevIdx = (currentTrackIdx - 1 + TRACKS.length) % TRACKS.length;

            isPlaying = true;

            loadTrack(prevIdx);

        }


        function playTrackFromPlaylist(idx) {

            isPlaying = true;

            loadTrack(idx);

        }


        // --- UI UPDATES ---

        function updatePlayStateUI() {

            mainPlayIcon.textContent = isPlaying ? 'pause' : 'play_arrow';

            

            if (isPlaying) {

                currentCoverContainer.classList.add('spin-slow');

            } else {

                currentCoverContainer.classList.remove('spin-slow');

            }

            

            renderStations(); 

            renderPlaylist(); 

        }


        function updateProgress() {

            const current = audio.currentTime || 0;

            const total = audio.duration || 0;

            

            timeCurrent.textContent = formatTime(current);

            timeTotal.textContent = formatTime(total);

            

            const percent = total > 0 ? (current / total) * 100 : 0;

            progressBar.style.width = `${percent}%`;

        }


        function seekTrack(e) {

            const rect = progressContainer.getBoundingClientRect();

            const percent = (e.clientX - rect.left) / rect.width;

            audio.currentTime = percent * (audio.duration || 0);

        }


        function changeVolume(val) {

            volume = parseFloat(val);

            audio.volume = volume;

            volumeBar.style.width = `${volume * 100}%`;

            

            if (volume === 0) {

                volumeIcon.textContent = 'volume_off';

            } else if (volume < 0.5) {

                volumeIcon.textContent = 'volume_down';

            } else {

                volumeIcon.textContent = 'volume_up';

            }

        }


        function toggleMute() {

            if (volume > 0) {

                volumeSlider.value = 0;

                changeVolume(0);

            } else {

                volumeSlider.value = 0.5;

                changeVolume(0.5);

            }

        }


        function togglePlaylist() {

            showPlaylist = !showPlaylist;

            if (showPlaylist) {

                playlistContainer.classList.remove('translate-x-full', 'hidden');

                playlistToggleBtn.classList.replace('text-white/60', 'text-white');

                playlistToggleBtn.classList.replace('hover:bg-white/10', 'bg-white/20');

            } else {

                playlistContainer.classList.add('translate-x-full', 'hidden');

                playlistToggleBtn.classList.replace('text-white', 'text-white/60');

                playlistToggleBtn.classList.replace('bg-white/20', 'hover:bg-white/10');

            }

        }


        function toggleFullscreen() {

            if (!document.fullscreenElement) {

                document.documentElement.requestFullscreen().catch(err => {

                    console.error(`Error enabling full-screen mode: ${err.message}`);

                });

            } else {

                document.exitFullscreen();

            }

        }

                    `;

        function renderPlaylist() {

            tracksContainer.innerHTML = TRACKS.map((track, idx) => {

                const isActive = currentTrackIdx === idx;

                

                let overlayIcon = '';

                if (isActive) {

                    overlayIcon = `

                        <div class="absolute inset-0 flex items-center justify-center bg-black/40">

                            <span class="material-symbols-rounded fill text-white" style="font-size: 20px;">

                                ${isPlaying ? 'pause' : 'play_arrow'}

                            </span>

                    `;

                }


                return `

                        onclick="playTrackFromPlaylist(${idx})"

                        class="flex items-center gap-3 p-2 rounded-lg cursor-pointer transition-all duration-200 group ${

                            isActive ? 'bg-white/10' : 'hover:bg-white/5'

                        }"

                    >

                        <div class="relative w-12 h-12 overflow-hidden rounded-md shadow-sm shrink-0">

        function formatTime(seconds) {

            if (!seconds || isNaN(seconds)) return "0:00";

            const m = Math.floor(seconds / 60);

            const s = Math.floor(seconds % 60);

            return `${m}:${s.toString().padStart(2, '0')}`;

        }
